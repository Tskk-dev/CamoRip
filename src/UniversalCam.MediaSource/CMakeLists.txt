cmake_minimum_required(VERSION 3.24)
project(UniversalCamMediaSource LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Use the Windows 11 SDK and MSVC x64 toolchain")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "UniversalCam v1 supports Windows x64 only")
endif()
add_library(UCamMediaSource SHARED MediaSource.cpp FrameChannel.cpp CameraHost.cpp UCamMediaSource.def)
target_compile_features(UCamMediaSource PRIVATE cxx_std_17)
target_compile_definitions(UCamMediaSource PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A00000B)
target_compile_options(UCamMediaSource PRIVATE /W4 /EHsc /permissive- /guard:cf)
target_link_options(UCamMediaSource PRIVATE /guard:cf)
set_property(TARGET UCamMediaSource PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(UCamMediaSource PRIVATE mfplat mf mfuuid mfsensorgroup ole32 advapi32 shlwapi propsys)
add_executable(ProviderSmokeTest ProviderSmokeTest.cpp)
target_compile_features(ProviderSmokeTest PRIVATE cxx_std_17)
target_compile_definitions(ProviderSmokeTest PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A00000B)
target_compile_options(ProviderSmokeTest PRIVATE /W4 /EHsc)
set_property(TARGET ProviderSmokeTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(ProviderSmokeTest PRIVATE mfplat mf mfuuid mfsensorgroup ole32 propsys)
enable_testing()
add_test(NAME ProviderSmokeTest COMMAND ProviderSmokeTest "$<TARGET_FILE:UCamMediaSource>")
set_tests_properties(ProviderSmokeTest PROPERTIES TIMEOUT 20)
add_executable(FrameChannelTests FrameChannelTests.cpp)
target_compile_features(FrameChannelTests PRIVATE cxx_std_17)
target_compile_definitions(FrameChannelTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A00000B)
target_compile_options(FrameChannelTests PRIVATE /W4 /EHsc)
set_property(TARGET FrameChannelTests PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(FrameChannelTests PRIVATE advapi32)
add_test(NAME FrameChannelTests COMMAND FrameChannelTests)
set_tests_properties(FrameChannelTests PROPERTIES TIMEOUT 10)
